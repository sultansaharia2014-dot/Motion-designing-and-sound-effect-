function clamp(value, min, max) {
  return Math.max(min, Math.min(max, value));
}

function safeFloor(value) {
  return Math.max(0.0001, value);
}

/** Deterministic Mulberry32 PRNG for reproducible noise, glitch bursts & reverb tails. */
export function createPrng(seed = 1) {
  let state = (seed >>> 0) || 1;
  return () => {
    state = (state + 0x6d2b79f5) | 0;
    let t = Math.imul(state ^ (state >>> 15), 1 | state);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

export function createAudioEventFromPreset(category, startTime = 0, overrides = {}) {
  const base = SOUND_PRESETS[category] || SOUND_PRESETS.ping;
  return {
    id: overrides.id || `sfx_${Math.random().toString(36).slice(2, 9)}_${Date.now().toString(36).slice(-4)}`,
    type: base.type,
    label: overrides.label || base.label,
    startTime: Math.max(0, Number(startTime) || 0),
    duration: Math.max(0.05, Number(overrides.duration ?? base.duration)),
    volume: clamp(Number(overrides.volume ?? base.volume), 0, 1),
    muted: Boolean(overrides.muted),
    solo: Boolean(overrides.solo),
    autoGenerated: Boolean(overrides.autoGenerated),
    sourceLayerId: overrides.sourceLayerId || null,
    parameters: {
      ...base.parameters,
      ...(overrides.parameters || {})
    }
  };
}

export function createNoiseBuffer(ctx, duration, seed = 42) {
  const sampleRate = ctx.sampleRate || 48000;
  const length = Math.max(1, Math.ceil(sampleRate * Math.max(0.05, duration)));
  const buffer = ctx.createBuffer(1, length, sampleRate);
  const channel = buffer.getChannelData(0);
  const prng = createPrng(seed);
  for (let i = 0; i < length; i++) {
    channel[i] = prng() * 2 - 1;
  }
  return buffer;
}

function makeSaturationCurve(amount = 16) {
  const samples = 256;
  const curve = new Float32Array(samples);
  const deg = Math.PI / 180;
  for (let i = 0; i < samples; i++) {
    const x = (i * 2) / samples - 1;
    curve[i] = ((3 + amount) * x * 20 * deg) / (Math.PI + amount * Math.abs(x));
  }
  return curve;
}

function connectWithOptionalPan(ctx, sourceNode, destination, panValue = 0) {
  if (typeof ctx.createStereoPanner === "function") {
    const panner = ctx.createStereoPanner();
    panner.pan.value = clamp(panValue, -1, 1);
    sourceNode.connect(panner);
    panner.connect(destination);
    return panner;
  }
  sourceNode.connect(destination);
  return null;
}

/** A. Digital Ping — Sine + optional triangle harmonic layer with crisp attack and clean tail. */
export function triggerPing(ctx, dest, event, when, masterGain = 1) {
  const dur = Math.max(0.08, event.duration || 0.25);
  const params = event.parameters || {};
  const intensity = clamp(params.intensity ?? 0.8, 0.1, 1);
  const peak = safeFloor((event.volume ?? 0.8) * masterGain * intensity * 0.45);
  const freq = Math.max(80, params.frequencyStart || 1046.5);
  const harmonic = Math.max(1, params.harmonic ?? 2);

  const oscPrimary = ctx.createOscillator();
  const oscHarmonic = ctx.createOscillator();
  const harmonicGain = ctx.createGain();
  const env = ctx.createGain();

  oscPrimary.type = "sine";
  oscPrimary.frequency.setValueAtTime(freq, when);

  oscHarmonic.type = "triangle";
  oscHarmonic.frequency.setValueAtTime(freq * harmonic, when);
  harmonicGain.gain.setValueAtTime(0.28, when);

  env.gain.setValueAtTime(0.0001, when);
  env.gain.linearRampToValueAtTime(peak, when + 0.008);
  env.gain.exponentialRampToValueAtTime(0.0001, when + dur);

  oscPrimary.connect(env);
  oscHarmonic.connect(harmonicGain);
  harmonicGain.connect(env);
  const panner = connectWithOptionalPan(ctx, env, dest, params.pan ?? 0);

  oscPrimary.start(when);
  oscHarmonic.start(when);
  oscPrimary.stop(when + dur + 0.025);
  oscHarmonic.stop(when + dur + 0.025);
  oscPrimary.onended = () => {
    oscPrimary.disconnect();
    oscHarmonic.disconnect();
    harmonicGain.disconnect();
    env.disconnect();
    panner?.disconnect();
  };
}

/** B. Whoosh — Seeded filtered noise burst with swept bandpass and stereo motion. */
export function triggerWhoosh(ctx, dest, event, when, masterGain = 1) {
  const dur = Math.max(0.14, event.duration || 0.45);
  const params = event.parameters || {};
  const intensity = clamp(params.intensity ?? 0.85, 0.1, 1);
  const peak = safeFloor((event.volume ?? 0.85) * masterGain * intensity * 0.6);
  const fStart = Math.max(60, params.frequencyStart || 140);
  const fEnd = Math.max(180, params.frequencyEnd || 2400);

  const src = ctx.createBufferSource();
  src.buffer = createNoiseBuffer(ctx, dur + 0.05, params.seed ?? 101);

  const filter = ctx.createBiquadFilter();
  filter.type = "bandpass";
  filter.Q.setValueAtTime(2.1, when);
  filter.frequency.setValueAtTime(fStart, when);
  filter.frequency.exponentialRampToValueAtTime(fEnd, when + dur * 0.52);
  filter.frequency.exponentialRampToValueAtTime(Math.max(70, fStart * 1.15), when + dur);

  const env = ctx.createGain();
  env.gain.setValueAtTime(0.0001, when);
  env.gain.linearRampToValueAtTime(peak, when + dur * 0.46);
  env.gain.exponentialRampToValueAtTime(0.0001, when + dur);

  src.connect(filter);
  filter.connect(env);

  let panner = null;
  if (typeof ctx.createStereoPanner === "function") {
    panner = ctx.createStereoPanner();
    const startPan = clamp(params.pan ?? -0.65, -1, 1);
    panner.pan.setValueAtTime(startPan, when);
    panner.pan.linearRampToValueAtTime(-startPan, when + dur);
    env.connect(panner);
    panner.connect(dest);
  } else {
    env.connect(dest);
  }

  src.start(when);
  src.stop(when + dur + 0.025);
  src.onended = () => {
    src.disconnect();
    filter.disconnect();
    env.disconnect();
    panner?.disconnect();
  };
}

/** C. Pop — Fast pitch-swept sine transient with optional crisp noise snap. */
export function triggerPop(ctx, dest, event, when, masterGain = 1) {
  const dur = clamp(event.duration || 0.14, 0.05, 0.3);
  const params = event.parameters || {};
  const intensity = clamp(params.intensity ?? 0.8, 0.1, 1);
  const peak = safeFloor((event.volume ?? 0.8) * masterGain * intensity * 0.58);
  const fStart = Math.max(110, params.frequencyStart || 560);
  const fEnd = Math.max(40, params.frequencyEnd || 135);

  const osc = ctx.createOscillator();
  const env = ctx.createGain();
  osc.type = "sine";
  osc.frequency.setValueAtTime(fStart, when);
  osc.frequency.exponentialRampToValueAtTime(fEnd, when + dur * 0.85);

  env.gain.setValueAtTime(0.0001, when);
  env.gain.linearRampToValueAtTime(peak, when + 0.004);
  env.gain.exponentialRampToValueAtTime(0.0001, when + dur);

  const noise = ctx.createBufferSource();
  noise.buffer = createNoiseBuffer(ctx, 0.03, params.seed ?? 33);
  const hp = ctx.createBiquadFilter();
  hp.type = "highpass";
  hp.frequency.setValueAtTime(1800, when);
  const nEnv = ctx.createGain();
  nEnv.gain.setValueAtTime(0.0001, when);
  nEnv.gain.linearRampToValueAtTime(peak * 0.22, when + 0.002);
  nEnv.gain.exponentialRampToValueAtTime(0.0001, when + 0.025);

  osc.connect(env);
  noise.connect(hp);
  hp.connect(nEnv);
  const panner = connectWithOptionalPan(ctx, env, dest, params.pan ?? 0);
  nEnv.connect(panner || dest);

  osc.start(when);
  noise.start(when);
  osc.stop(when + dur + 0.02);
  noise.stop(when + 0.035);
  osc.onended = () => {
    osc.disconnect();
    env.disconnect();
    noise.disconnect();
    hp.disconnect();
    nEnv.disconnect();
    panner?.disconnect();
  };
}

/** D. Bass Drop — Pitch-descending sine sub + softly saturated triangle body + LP sweep. */
export function triggerBassDrop(ctx, dest, event, when, masterGain = 1) {
  const dur = Math.max(0.32, event.duration || 0.95);
  const params = event.parameters || {};
  const intensity = clamp(params.intensity ?? 0.95, 0.1, 1);
  const peak = safeFloor((event.volume ?? 0.9) * masterGain * intensity * 0.72);
  const fStart = Math.max(55, params.frequencyStart || 148);
  const fEnd = Math.max(26, params.frequencyEnd || 34);

  const sub = ctx.createOscillator();
  const body = ctx.createOscillator();
  const shaper = ctx.createWaveShaper();
  const lp = ctx.createBiquadFilter();
  const env = ctx.createGain();

  sub.type = "sine";
  sub.frequency.setValueAtTime(fStart, when);
  sub.frequency.exponentialRampToValueAtTime(fEnd, when + dur * 0.62);

  body.type = "triangle";
  body.frequency.setValueAtTime(fStart * (params.harmonic ?? 1.5), when);
  body.frequency.exponentialRampToValueAtTime(fEnd * 2, when + dur * 0.52);

  shaper.curve = makeSaturationCurve(params.distortion ?? 16);
  lp.type = "lowpass";
  lp.frequency.setValueAtTime(480, when);
  lp.frequency.exponentialRampToValueAtTime(95, when + dur);

  env.gain.setValueAtTime(0.0001, when);
  env.gain.linearRampToValueAtTime(peak, when + 0.014);
  env.gain.setTargetAtTime(0.0001, when + 0.06, dur * 0.28);

  sub.connect(env);
  body.connect(shaper);
  shaper.connect(lp);
  lp.connect(env);
  const panner = connectWithOptionalPan(ctx, env, dest, params.pan ?? 0);

  sub.start(when);
  body.start(when);
  sub.stop(when + dur + 0.06);
  body.stop(when + dur + 0.06);
  sub.onended = () => {
    sub.disconnect();
    body.disconnect();
    shaper.disconnect();
    lp.disconnect();
    env.disconnect();
    panner?.disconnect();
  };
}

/** E. Cinematic Impact — Sub transient + filtered noise + metallic overtone + procedural reverb tail. */
export function triggerImpact(ctx, dest, event, when, masterGain = 1, reverbNode = null) {
  const dur = Math.max(0.2, event.duration || 0.5);
  const params = event.parameters || {};
  const intensity = clamp(params.intensity ?? 0.95, 0.1, 1);
  const peak = safeFloor((event.volume ?? 0.9) * masterGain * intensity * 0.7);
  const fStart = Math.max(70, params.frequencyStart || 175);
  const fEnd = Math.max(28, params.frequencyEnd || 38);

  const sub = ctx.createOscillator();
  const subEnv = ctx.createGain();
  sub.type = "sine";
  sub.frequency.setValueAtTime(fStart, when);
  sub.frequency.exponentialRampToValueAtTime(fEnd, when + Math.min(0.24, dur));
  subEnv.gain.setValueAtTime(0.0001, when);
  subEnv.gain.linearRampToValueAtTime(peak, when + 0.006);
  subEnv.gain.exponentialRampToValueAtTime(0.0001, when + dur);

  const noise = ctx.createBufferSource();
  noise.buffer = createNoiseBuffer(ctx, dur, params.seed ?? 77);
  const lp = ctx.createBiquadFilter();
  lp.type = "lowpass";
  lp.frequency.setValueAtTime(3200, when);
  lp.frequency.exponentialRampToValueAtTime(190, when + dur * 0.85);
  const nEnv = ctx.createGain();
  nEnv.gain.setValueAtTime(0.0001, when);
  nEnv.gain.linearRampToValueAtTime(peak * 0.55, when + 0.004);
  nEnv.gain.exponentialRampToValueAtTime(0.0001, when + dur * 0.78);

  // Optional metallic resonance ring
  const metalAmt = clamp(params.metallic ?? 0.3, 0, 1);
  const metalOsc = ctx.createOscillator();
  const metalEnv = ctx.createGain();
  metalOsc.type = "triangle";
  metalOsc.frequency.setValueAtTime(432, when);
  metalOsc.frequency.exponentialRampToValueAtTime(216, when + dur * 0.4);
  metalEnv.gain.setValueAtTime(0.0001, when);
  metalEnv.gain.linearRampToValueAtTime(safeFloor(peak * metalAmt * 0.35), when + 0.005);
  metalEnv.gain.exponentialRampToValueAtTime(0.0001, when + dur * 0.45);

  const panner = connectWithOptionalPan(ctx, subEnv, dest, params.pan ?? 0);
  noise.connect(lp);
  lp.connect(nEnv);
  nEnv.connect(panner || dest);
  metalOsc.connect(metalEnv);
  metalEnv.connect(panner || dest);

  if (reverbNode) {
    nEnv.connect(reverbNode);
    metalEnv.connect(reverbNode);
  }

  sub.start(when);
  noise.start(when);
  metalOsc.start(when);
  sub.stop(when + dur + 0.03);
  noise.stop(when + dur + 0.03);
  metalOsc.stop(when + dur + 0.03);
  sub.onended = () => {
    sub.disconnect();
    subEnv.disconnect();
    noise.disconnect();
    lp.disconnect();
    nEnv.disconnect();
    metalOsc.disconnect();
    metalEnv.disconnect();
    panner?.disconnect();
  };
}

/** F. Glitch — Seeded quantized pitch stepping, amplitude chop, bandpass jitter & optional micro-delay. */
export function triggerGlitch(ctx, dest, event, when, masterGain = 1) {
  const dur = Math.max(0.1, event.duration || 0.3);
  const params = event.parameters || {};
  const intensity = clamp(params.intensity ?? 0.85, 0.1, 1);
  const peak = safeFloor((event.volume ?? 0.75) * masterGain * intensity * 0.45);
  const prng = createPrng(params.seed ?? 909);

  const osc = ctx.createOscillator();
  const filter = ctx.createBiquadFilter();
  const env = ctx.createGain();
  osc.type = "square";
  filter.type = "bandpass";
  filter.Q.setValueAtTime(4.5, when);

  const steps = Math.max(4, Math.floor(dur / 0.028));
  const stepDur = dur / steps;
  const fMin = Math.max(90, params.frequencyStart || 200);
  const fMax = Math.max(fMin + 200, params.frequencyEnd || 2800);

  env.gain.setValueAtTime(0.0001, when);
  for (let i = 0; i < steps; i++) {
    const t = when + i * stepDur;
    osc.frequency.setValueAtTime(fMin + prng() * (fMax - fMin), t);
    filter.frequency.setValueAtTime(320 + prng() * 3400, t);
    env.gain.setValueAtTime(prng() > 0.32 ? peak : 0.0001, t);
  }
  env.gain.linearRampToValueAtTime(0.0001, when + dur);

  osc.connect(filter);
  filter.connect(env);
  const panner = connectWithOptionalPan(ctx, env, dest, params.pan ?? 0);

  let delay = null;
  let fb = null;
  if (typeof ctx.createDelay === "function" && (params.echo ?? 0.25) > 0.02) {
    delay = ctx.createDelay(0.25);
    delay.delayTime.setValueAtTime(0.045, when);
    fb = ctx.createGain();
    fb.gain.setValueAtTime(clamp(params.echo ?? 0.25, 0, 0.55), when);
    env.connect(delay);
    delay.connect(fb);
    fb.connect(dest);
  }

  osc.start(when);
  osc.stop(when + dur + 0.03);
  osc.onended = () => {
    osc.disconnect();
    filter.disconnect();
    env.disconnect();
    panner?.disconnect();
    delay?.disconnect();
    fb?.disconnect();
  };
}

/** G. Riser — Continuous upward pitch & filter sweep blending sawtooth oscillator and noise. */
export function triggerRiser(ctx, dest, event, when, masterGain = 1) {
  const dur = Math.max(0.35, event.duration || 1.25);
  const params = event.parameters || {};
  const intensity = clamp(params.intensity ?? 0.88, 0.1, 1);
  const peak = safeFloor((event.volume ?? 0.82) * masterGain * intensity * 0.5);
  const fStart = Math.max(70, params.frequencyStart || 120);
  const fEnd = Math.max(fStart * 2, params.frequencyEnd || 1950);

  const osc = ctx.createOscillator();
  osc.type = "sawtooth";
  osc.frequency.setValueAtTime(fStart, when);
  osc.frequency.exponentialRampToValueAtTime(fEnd, when + dur);

  const noise = ctx.createBufferSource();
  noise.buffer = createNoiseBuffer(ctx, dur + 0.05, params.seed ?? 303);

  const filter = ctx.createBiquadFilter();
  filter.type = "bandpass";
  filter.Q.setValueAtTime(3.0, when);
  filter.frequency.setValueAtTime(fStart * 1.4, when);
  filter.frequency.exponentialRampToValueAtTime(fEnd * 1.8, when + dur);

  const env = ctx.createGain();
  env.gain.setValueAtTime(0.0001, when);
  env.gain.exponentialRampToValueAtTime(peak, when + dur * 0.92);
  env.gain.linearRampToValueAtTime(0.0001, when + dur);

  osc.connect(filter);
  noise.connect(filter);
  filter.connect(env);
  const panner = connectWithOptionalPan(ctx, env, dest, params.pan ?? 0);

  osc.start(when);
  noise.start(when);
  osc.stop(when + dur + 0.025);
  noise.stop(when + dur + 0.025);
  osc.onended = () => {
    osc.disconnect();
    noise.disconnect();
    filter.disconnect();
    env.disconnect();
    panner?.disconnect();
  };
}

export class ProceduralSynthesizer {
  constructor() {
    this.reverbCache = new WeakMap();
  }

  getReverbNode(ctx, dest) {
    if (typeof ctx.createConvolver !== "function") return null;
    const cached = this.reverbCache.get(ctx);
    if (cached) return cached;

    const sampleRate = ctx.sampleRate || 48000;
    const length = Math.max(1, Math.floor(sampleRate * 0.45));
    const impulse = ctx.createBuffer(2, length, sampleRate);
    const prng = createPrng(5150);
    for (let ch = 0; ch < 2; ch++) {
      const data = impulse.getChannelData(ch);
      for (let i = 0; i < length; i++) {
        data[i] = (prng() * 2 - 1) * Math.pow(1 - i / length, 2.5);
      }
    }
    const convolver = ctx.createConvolver();
    convolver.buffer = impulse;
    const wetGain = ctx.createGain();
    wetGain.gain.value = 0.24;
    convolver.connect(wetGain);
    wetGain.connect(dest);
    this.reverbCache.set(ctx, convolver);
    return convolver;
  }

  triggerEvent(ctx, dest, event, when, masterGain = 1) {
    const t = Math.max(0.001, when);
    switch (event.type) {
      case "ping":
        return triggerPing(ctx, dest, event, t, masterGain);
      case "whoosh":
        return triggerWhoosh(ctx, dest, event, t, masterGain);
      case "pop":
        return triggerPop(ctx, dest, event, t, masterGain);
      case "bassDrop":
        return triggerBassDrop(ctx, dest, event, t, masterGain);
      case "impact":
        return triggerImpact(ctx, dest, event, t, masterGain, this.getReverbNode(ctx, dest));
      case "glitch":
        return triggerGlitch(ctx, dest, event, t, masterGain);
      case "riser":
        return triggerRiser(ctx, dest, event, t, masterGain);
      default:
        return triggerPing(ctx, dest, event, t, masterGain);
    }
  }
}

export class AudioScheduler {
  /** Maps visual animation events (text hits, transitions, HUD pings, chibi pops) to SFX. */
  static autoMapVisualEvents(project) {
    const manual = (project.audioEvents || []).filter((e) => !e.autoGenerated);
    if (project.settings && project.settings.autoSfx === false) return manual;

    const generated = [];
    for (const layer of project.layers || []) {
      if (layer.visible === false) continue;
      if (layer.type === "text") {
        generated.push(
          createAudioEventFromPreset("impact", layer.startTime || 0, {
            label: `Text Impact: ${layer.name || layer.id}`,
            autoGenerated: true,
            sourceLayerId: layer.id
          })
        );
      } else if (layer.type === "hud" || layer.type === "infographic") {
        generated.push(
          createAudioEventFromPreset("ping", layer.startTime || 0, {
            label: `UI Ping: ${layer.name || layer.id}`,
            autoGenerated: true,
            sourceLayerId: layer.id
          })
        );
      } else if (layer.type === "character2d" || layer.type === "particle") {
        generated.push(
          createAudioEventFromPreset("pop", layer.startTime || 0, {
            label: `Pop: ${layer.name || layer.id}`,
            autoGenerated: true,
            sourceLayerId: layer.id
          })
        );
      } else if (layer.type === "three3d") {
        generated.push(
          createAudioEventFromPreset("bassDrop", layer.startTime || 0, {
            label: `3D Reveal: ${layer.name || layer.id}`,
            volume: 0.72,
            autoGenerated: true,
            sourceLayerId: layer.id
          })
        );
      }
    }

    for (const scene of project.scenes || []) {
      if (!scene.transition || scene.transition === "none") continue;
      const tDur = Math.max(0.2, scene.transitionDuration || 0.35);
      const tStart = Math.max(0, (scene.startTime || 0) + (scene.duration || 0) - tDur * 0.5);
      const kind = scene.transition === "glitchCut"
        ? "glitch"
        : scene.transition === "zoomPunch"
          ? "bassDrop"
          : "whoosh";
      generated.push(
        createAudioEventFromPreset(kind, tStart, {
          label: `Scene Sweep: ${scene.name || scene.id}`,
          duration: tDur,
          autoGenerated: true
        })
      );
    }

    return [...manual, ...generated].sort((a, b) => a.startTime - b.startTime);
  }

  static resolvePlayable(events) {
    const list = Array.isArray(events) ? events : [];
    const hasSolo = list.some((e) => Boolean(e.solo));
    return list.filter((e) => !e.muted && (!hasSolo || Boolean(e.solo)));
  }
}

export class AudioMixer {
  constructor(initialVolume = 0.85) {
    this.masterVolume = clamp(initialVolume, 0, 1);
    this.masterMuted = false;
  }

  effectiveGain() {
    return this.masterMuted ? 0 : this.masterVolume;
  }
}

export class AudioTimeline {
  constructor() {
    this.firedIds = new Set();
  }

  reset() {
    this.firedIds.clear();
  }

  collectCrossedEvents(events, prevTime, currentTime) {
    if (currentTime < prevTime) this.firedIds.clear();
    const playable = AudioScheduler.resolvePlayable(events);
    const due = [];
    for (const ev of playable) {
      const hit = ev.startTime >= prevTime - 0.02 && ev.startTime <= currentTime + 0.02;
      if (hit && !this.firedIds.has(ev.id)) {
        this.firedIds.add(ev.id);
        due.push(ev);
      }
    }
    return due;
  }
}

export function audioBufferToWavBlob(buffer) {
  const numChannels = Math.min(2, buffer.numberOfChannels || 1);
  const sampleRate = buffer.sampleRate || 48000;
  const length = buffer.length;
  const dataSize = length * numChannels * 2;
  const view = new DataView(new ArrayBuffer(44 + dataSize));
  const writeStr = (offset, str) => {
    for (let i = 0; i < str.length; i++) view.setUint8(offset + i, str.charCodeAt(i));
  };

  writeStr(0, "RIFF");
  view.setUint32(4, 36 + dataSize, true);
  writeStr(8, "WAVEfmt ");
  view.setUint32(16, 16, true);
  view.setUint16(20, 1, true);
  view.setUint16(22, numChannels, true);
  view.setUint32(24, sampleRate, true);
  view.setUint32(28, sampleRate * numChannels * 2, true);
  view.setUint16(32, numChannels * 2, true);
  view.setUint16(34, 16, true);
  writeStr(36, "data");
  view.setUint32(40, dataSize, true);

  let offset = 44;
  for (let i = 0; i < length; i++) {
    for (let ch = 0; ch < numChannels; ch++) {
      const sample = clamp(buffer.getChannelData(ch)[i] || 0, -1, 1);
      view.setInt16(offset, sample < 0 ? sample * 0x8000 : sample * 0x7fff, true);
      offset += 2;
    }
  }
  return new Blob([view.buffer], { type: "audio/wav" });
}

export class AudioEngine {
  constructor(AudioContextImpl = null, OfflineContextImpl = null) {
    this.AudioContextImpl = AudioContextImpl;
    this.OfflineContextImpl = OfflineContextImpl;
    this.ctx = null;
    this.masterGainNode = null;
    this.exportDestNode = null;
    this.synthesizer = new ProceduralSynthesizer();
    this.mixer = new AudioMixer(0.85);
    this.timeline = new AudioTimeline();
  }

  /** Must be triggered from an explicit user interaction when autoplay restrictions apply. */
  async ensureInitialized() {
    if (!this.ctx) {
      const Ctx =
        this.AudioContextImpl ||
        (typeof window !== "undefined" && (window.AudioContext || window.webkitAudioContext));
      if (!Ctx) throw new Error("Web Audio API AudioContext is unavailable in this environment.");
      this.ctx = new Ctx();
      this.masterGainNode = this.ctx.createGain();
      this.masterGainNode.gain.value = this.mixer.effectiveGain();
      this.masterGainNode.connect(this.ctx.destination);
    }
    if (this.ctx.state === "suspended" && typeof this.ctx.resume === "function") {
      await this.ctx.resume();
    }
    return this.ctx;
  }

  setMasterVolume(vol) {
    this.mixer.masterVolume = clamp(vol, 0, 1);
    if (this.masterGainNode && this.ctx) {
      this.masterGainNode.gain.setTargetAtTime(this.mixer.effectiveGain(), this.ctx.currentTime, 0.015);
    }
  }

  setMasterMuted(muted) {
    this.mixer.masterMuted = Boolean(muted);
    this.setMasterVolume(this.mixer.masterVolume);
  }

  resetTriggerWindow() {
    this.timeline.reset();
  }

  async previewEvent(event) {
    const ctx = await this.ensureInitialized();
    const when = ctx.currentTime + 0.01;
    this.synthesizer.triggerEvent(ctx, this.masterGainNode, event, when, 1);
  }

  syncPlaybackTime(events, prevTime, currentTime) {
    if (!this.ctx || !this.masterGainNode || this.ctx.state !== "running") return;
    const due = this.timeline.collectCrossedEvents(events, prevTime, currentTime);
    for (const ev of due) {
      this.synthesizer.triggerEvent(this.ctx, this.masterGainNode, ev, this.ctx.currentTime + 0.005, 1);
    }
  }

  /** Creates a MediaStreamAudioDestinationNode capturing the live master mix for MediaRecorder. */
  async attachExportStreamDestination() {
    const ctx = await this.ensureInitialized();
    if (!this.exportDestNode && typeof ctx.createMediaStreamDestination === "function") {
      this.exportDestNode = ctx.createMediaStreamDestination();
      this.masterGainNode.connect(this.exportDestNode);
    }
    return this.exportDestNode ? this.exportDestNode.stream : null;
  }

  /** Renders an entire deterministic AudioBuffer offline for frame-synced export or WAV download. */
  async renderOfflineMix(events, durationSeconds, sampleRate = 48000) {
    const OfflineCtx =
      this.OfflineContextImpl ||
      (typeof window !== "undefined" && (window.OfflineAudioContext || window.webkitOfflineAudioContext));
    if (!OfflineCtx) {
      throw new Error("OfflineAudioContext is not supported in this environment.");
    }
    const dur = Math.max(0.1, durationSeconds);
    const frameCount = Math.ceil(dur * sampleRate);
    const offline = new OfflineCtx(2, frameCount, sampleRate);
    const master = offline.createGain();
    master.gain.value = this.mixer.effectiveGain();
    master.connect(offline.destination);

    const playable = AudioScheduler.resolvePlayable(events);
    for (const ev of playable) {
      if (ev.startTime >= 0 && ev.startTime < dur) {
        this.synthesizer.triggerEvent(offline, master, ev, Math.max(0.002, ev.startTime), 1);
      }
    }
    return offline.startRendering();
  }

  dispose() {
    if (this.ctx && this.ctx.state !== "closed" && typeof this.ctx.close === "function") {
      void this.ctx.close();
    }
    this.ctx = null;
    this.masterGainNode = null;
    this.exportDestNode = null;
    this.timeline.reset();
  }
}
